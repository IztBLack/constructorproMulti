// Prototipo de `PdfService.checklistMaterial`: mismo encabezado, paleta, título
// de sección y firmas que lib/pdf/pdf_service.dart, con una casilla por renglón.
import 'dart:io';

import 'package:pdf/pdf.dart';
import 'package:pdf/widgets.dart' as pw;

const _slate = PdfColor.fromInt(0xFF0F172A);
const _gris700 = PdfColor.fromInt(0xFF404040);
const _gris500 = PdfColor.fromInt(0xFF737373);
const _gris400 = PdfColor.fromInt(0xFFA3A3A3);
const _gris200 = PdfColor.fromInt(0xFFE5E5E5);
const _gris100 = PdfColor.fromInt(0xFFF1F1F1);
const _grisFondo = PdfColor.fromInt(0xFFFAFAFA);
const _acento = PdfColor.fromInt(0xFF1A3A5C);

class Pieza {
  final int cantidad;
  final double largo;
  final String nota;
  const Pieza(this.cantidad, this.largo, [this.nota = '']);
  double get metros => cantidad * largo;
}

class Seccion {
  final String nombre;
  final List<Pieza> piezas;
  const Seccion(this.nombre, this.piezas);
  int get piezasTotal => piezas.fold(0, (a, p) => a + p.cantidad);
  double get metrosTotal => piezas.fold(0.0, (a, p) => a + p.metros);
}

/// Estribos a separación fija: uno al inicio y uno por cada tramo completo.
int estribosPorPieza(double largo, double sepCm) =>
    (largo * 100 / sepCm).floor() + 1;

String m(double v) => v.toStringAsFixed(2);

pw.Widget casilla() => pw.Container(
      width: 11,
      height: 11,
      decoration: pw.BoxDecoration(
        border: pw.Border.all(color: _slate, width: 1),
        borderRadius: pw.BorderRadius.circular(2),
      ),
    );

pw.Widget celda(String t,
        {bool bold = false,
        pw.Alignment align = pw.Alignment.centerLeft,
        PdfColor color = _gris700,
        double size = 10}) =>
    pw.Container(
      alignment: align,
      padding: const pw.EdgeInsets.symmetric(horizontal: 6, vertical: 5),
      child: pw.Text(t,
          style: pw.TextStyle(
              fontSize: size,
              color: color,
              fontWeight: bold ? pw.FontWeight.bold : pw.FontWeight.normal)),
    );

pw.Widget header(String titulo, String subtitulo) => pw.Column(
      crossAxisAlignment: pw.CrossAxisAlignment.start,
      children: [
        pw.Container(
          padding: const pw.EdgeInsets.only(bottom: 12),
          decoration: const pw.BoxDecoration(
            border: pw.Border(bottom: pw.BorderSide(color: _slate, width: 2)),
          ),
          child: pw.Row(
            crossAxisAlignment: pw.CrossAxisAlignment.start,
            children: [
              pw.Expanded(
                child: pw.Column(
                  crossAxisAlignment: pw.CrossAxisAlignment.start,
                  children: [
                    pw.Text(titulo.toUpperCase(),
                        style: pw.TextStyle(
                            color: _acento,
                            fontSize: 9,
                            fontWeight: pw.FontWeight.bold,
                            letterSpacing: 1.2)),
                    pw.SizedBox(height: 3),
                    pw.Text('ConstructorPro',
                        style: pw.TextStyle(
                            color: _slate,
                            fontSize: 18,
                            fontWeight: pw.FontWeight.bold)),
                  ],
                ),
              ),
              pw.Padding(
                padding: const pw.EdgeInsets.only(left: 16),
                child: pw.Text(subtitulo,
                    textAlign: pw.TextAlign.right,
                    style: const pw.TextStyle(fontSize: 10, color: _gris700)),
              ),
            ],
          ),
        ),
        pw.SizedBox(height: 16),
      ],
    );

pw.Widget sectionTitle(String texto) => pw.Container(
      margin: const pw.EdgeInsets.only(bottom: 6),
      padding: const pw.EdgeInsets.only(left: 8),
      decoration: const pw.BoxDecoration(
        border: pw.Border(left: pw.BorderSide(color: _acento, width: 3)),
      ),
      child: pw.Text(texto.toUpperCase(),
          style: pw.TextStyle(
              fontSize: 10,
              fontWeight: pw.FontWeight.bold,
              color: _slate,
              letterSpacing: 0.5)),
    );

pw.Widget tablaSeccion(Seccion s) {
  const borde = pw.BorderSide(color: _gris100);
  return pw.Table(
    columnWidths: const {
      0: pw.FixedColumnWidth(30),
      1: pw.FixedColumnWidth(50),
      2: pw.FixedColumnWidth(70),
      3: pw.FlexColumnWidth(4),
      4: pw.FixedColumnWidth(80),
    },
    border: const pw.TableBorder(horizontalInside: borde, bottom: borde),
    children: [
      pw.TableRow(
        decoration: const pw.BoxDecoration(color: _grisFondo),
        children: [
          celda('OK', bold: true, size: 8, color: _gris500, align: pw.Alignment.center),
          celda('CANT.', bold: true, size: 8, color: _gris500, align: pw.Alignment.centerRight),
          celda('LARGO (m)', bold: true, size: 8, color: _gris500, align: pw.Alignment.centerRight),
          celda('NOTA', bold: true, size: 8, color: _gris500),
          celda('METROS LIN.', bold: true, size: 8, color: _gris500, align: pw.Alignment.centerRight),
        ],
      ),
      for (final p in s.piezas)
        pw.TableRow(
          verticalAlignment: pw.TableCellVerticalAlignment.middle,
          children: [
            pw.Center(child: casilla()),
            celda('${p.cantidad}', bold: true, color: _slate, align: pw.Alignment.centerRight),
            celda(m(p.largo), align: pw.Alignment.centerRight),
            celda(p.nota),
            celda(m(p.metros), align: pw.Alignment.centerRight),
          ],
        ),
      pw.TableRow(
        children: [
          pw.SizedBox(),
          celda('${s.piezasTotal}', bold: true, color: _slate, align: pw.Alignment.centerRight),
          celda('piezas', size: 8, color: _gris500),
          pw.SizedBox(),
          celda('${m(s.metrosTotal)} m', bold: true, color: _slate, align: pw.Alignment.centerRight),
        ],
      ),
    ],
  );
}

pw.Widget totalLinea(String label, String valor, {bool bold = false}) =>
    pw.Align(
      alignment: pw.Alignment.centerRight,
      child: pw.Container(
        width: 300,
        padding: const pw.EdgeInsets.symmetric(vertical: 6),
        decoration: pw.BoxDecoration(
          border: bold
              ? const pw.Border(top: pw.BorderSide(color: _slate, width: 2))
              : const pw.Border(bottom: pw.BorderSide(color: _gris100)),
        ),
        child: pw.Row(
          mainAxisAlignment: pw.MainAxisAlignment.spaceBetween,
          children: [
            pw.Text(label,
                style: pw.TextStyle(
                    fontSize: bold ? 13 : 11,
                    fontWeight: bold ? pw.FontWeight.bold : pw.FontWeight.normal,
                    color: bold ? _slate : _gris500)),
            pw.Text(valor,
                style: pw.TextStyle(
                    fontSize: bold ? 13 : 11,
                    fontWeight: bold ? pw.FontWeight.bold : pw.FontWeight.normal,
                    color: _slate)),
          ],
        ),
      ),
    );

pw.Widget firmas() => pw.Padding(
      padding: const pw.EdgeInsets.only(top: 30),
      child: pw.Row(
        mainAxisAlignment: pw.MainAxisAlignment.spaceAround,
        children: [
          for (final t in ['Elaboró', 'Revisó / recibió'])
            pw.Column(children: [
              pw.Container(width: 180, height: 1, color: PdfColors.black),
              pw.SizedBox(height: 4),
              pw.Text(t, style: const pw.TextStyle(fontSize: 10)),
            ]),
        ],
      ),
    );

Future<void> main(List<String> args) async {
  final salida = args.isNotEmpty ? args.first : 'checklist.pdf';
  const sepEstriboCm = 25.0;

  final secciones = [
    const Seccion('Trabes', [
      Pieza(4, 10.45),
      Pieza(2, 12.25),
      Pieza(2, 2.80),
      Pieza(1, 2.80, 'Chica'),
      Pieza(2, 3.90),
      Pieza(2, 3.10),
      Pieza(2, 2.10),
    ]),
    Seccion('Pasillo', [
      Pieza(2, 3.55,
          'Estribo @25 cm (~${estribosPorPieza(3.55, sepEstriboCm)} por pieza)'),
      const Pieza(1, 4.10),
      const Pieza(2, 3.00),
    ]),
  ];

  final piezas = secciones.fold(0, (a, s) => a + s.piezasTotal);
  final metros = secciones.fold(0.0, (a, s) => a + s.metrosTotal);
  final hoy = DateTime.now();
  final fecha =
      '${hoy.day.toString().padLeft(2, '0')}/${hoy.month.toString().padLeft(2, '0')}/${hoy.year}';

  final doc = pw.Document();
  doc.addPage(pw.MultiPage(
    pageTheme: const pw.PageTheme(
      pageFormat: PdfPageFormat.letter,
      margin: pw.EdgeInsets.symmetric(horizontal: 51, vertical: 45),
    ),
    footer: (ctx) => pw.Container(
      alignment: pw.Alignment.center,
      margin: const pw.EdgeInsets.only(top: 10),
      padding: const pw.EdgeInsets.only(top: 8),
      decoration: const pw.BoxDecoration(
        border: pw.Border(top: pw.BorderSide(color: _gris200)),
      ),
      child: pw.Text('Hoja ${ctx.pageNumber} de ${ctx.pagesCount}',
          style: const pw.TextStyle(fontSize: 9, color: _gris400)),
    ),
    build: (ctx) => [
      header(
        'Lista de material',
        [
          'Obra: ____________________',
          'Fecha: $fecha',
          'Medidas en metros',
        ].join('\n'),
      ),
      for (final s in secciones) ...[
        sectionTitle(s.nombre),
        tablaSeccion(s),
        pw.SizedBox(height: 18),
      ],
      for (final s in secciones)
        totalLinea(s.nombre, '${s.piezasTotal} pzas · ${m(s.metrosTotal)} m'),
      totalLinea('TOTAL', '$piezas pzas · ${m(metros)} m', bold: true),
      pw.SizedBox(height: 16),
      sectionTitle('Observaciones'),
      for (var i = 0; i < 3; i++)
        pw.Container(
          height: 22,
          decoration: const pw.BoxDecoration(
            border: pw.Border(bottom: pw.BorderSide(color: _gris200)),
          ),
        ),
      firmas(),
    ],
  ));

  await File(salida).writeAsBytes(await doc.save());
  stdout.writeln('OK $salida  ($piezas piezas, ${m(metros)} m)');
}
